# Run from the repository root. This registration has one capture only.
set -euo pipefail
plan_root="$PWD/evals/paired/agent-maintenance-v7-snapshot-retry-screen-2026-11-25"
shasum -a 256 -c "$plan_root/freeze-manifest.sha256" >/dev/null
capture_dir="$plan_root/capture-01"
if test -e "$capture_dir"; then
  printf 'Refusing to overwrite existing capture directory: %s\n' "$capture_dir" >&2
  exit 1
fi

private_root=$(mktemp -d /tmp/a11-v7-codex-home.XXXXXX)
trap 'rm -rf "$private_root"' EXIT
umask 077
mkdir -m 700 -p "$private_root/home" "$private_root/codex-home"
cp "$HOME/.codex/auth.json" "$private_root/codex-home/auth.json"
chmod 600 "$private_root/codex-home/auth.json"

target/debug/memory-eval run-paired-codex \
  --plan "$plan_root/plan.json" \
  --tasks "$plan_root/tasks.json" \
  --codex-binary "/Applications/ChatGPT.app/Contents/Resources/codex-cli/CodexCLI.app/Contents/MacOS/codex" \
  --codex-home "$private_root/codex-home" \
  --memory-tool-profile agent_maintenance_v7 \
  --incumbent-memory-server "$PWD/target/debug/memory_mcp" \
  --candidate-memory-server "$PWD/target/debug/memory_mcp" \
  --output-directory "$capture_dir"

target/debug/memory-eval compare-agent-managed \
  --plan "$plan_root/plan.json" \
  --incumbent "$capture_dir/incumbent.json" \
  --candidate "$capture_dir/candidate.json" \
  --incumbent-stages "$capture_dir/agent-managed-incumbent.json" \
  --candidate-stages "$capture_dir/agent-managed-candidate.json" \
  --output "$capture_dir/agent-managed-comparison.json"
